binary_search.py 
def binary_search(arr, target):
    left, right = 0, len(arr) - 1
    
    while left <= right:
        mid = (left + right) // 2
        
        if arr[mid] == target:
            return mid  # Found it! 🎯
        elif arr[mid] < target:
            left = mid + 1
        else:
            right = mid - 1
    
    return -1
